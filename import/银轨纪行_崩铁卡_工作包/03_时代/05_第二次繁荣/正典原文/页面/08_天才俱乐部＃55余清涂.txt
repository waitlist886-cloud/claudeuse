        $('.resourceLoader').each(function () {
            var $x = $(this);
            var text = $.trim($x.text());
          
            if (!text) return;
          
            //加载模块
            if ($x.data('isModule') === true)
                return mw.loader.load(text);

            var ns = text.match('^.*?:');
            if (!ns) text = 'MediaWiki:' + text;

            if (mime == "text/css") {
